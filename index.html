1. Kết nối Gemini AI (Đọc nhiều ảnh trang sách & Chấm chữ viết tay)

Lấy mã miễn phí tại aistudio.google.com (mục Get API key) và dán vào ô dưới đây. Mã được lưu an toàn trong LocalStorage của trình duyệt.

2. Đồng bộ 2 chiều qua Google Sheets (Bài học, Mốc Video & Điểm số)

Dán đường link Google Apps Script Web App URL để đồng bộ cả danh sách bài học (kèm link video + mốc thời gian từng đoạn) và lịch sử điểm số giữa máy tính của con và điện thoại của mẹ.

Hướng dẫn cài đặt Google Sheets trong 2 phút:

  1. Mở Google Drive và tạo một file Google Trang tính (Google Sheets) mới.
  2. Trên thanh menu chọn Tiện ích mở rộng (Extensions) → Apps Script.
  3. Bấm nút "Sao chép mã Apps Script" bên dưới, dán đè toàn bộ vào trình soạn thảo Apps Script.
  4. Bấm Triển khai (Deploy) → Tùy chọn triển khai mới → Ứng dụng web (Web app), chọn mục Ai có quyền truy cập là Bất kỳ ai (Anyone) rồi bấm Triển khai.
  5. Copy đường link Web App dán vào ô phía trên trên cả máy tính và điện thoại rồi bấm Lưu cấu hình.

Mã Google Apps Script (Hỗ trợ 2 sheet: Lessons & Scores): 

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);

  if (payload.action === "saveScore") {
    var sheet = ss.getSheetByName("Scores") || ss.insertSheet("Scores");
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "DateISO", "DateVN", "LessonTitle", "Scope", "Mode", "Correct", "Total", "Percent", "WrongWords"]);
    }
    var r = payload.data;
    sheet.appendRow([
      r.id, r.dateISO, r.dateVN, r.lessonTitle, r.scopeLabel,
      r.modeLabel, r.correct, r.total, r.percent, (r.wrongWords || []).join(", ")
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "saveLesson") {
    var lSheet = ss.getSheetByName("Lessons") || ss.insertSheet("Lessons");
    if (lSheet.getLastRow() === 0) {
      lSheet.appendRow(["ID", "Title", "MediaUrl", "ParagraphsJSON", "TimestampsJSON", "KeywordsJSON"]);
    }
    var l = payload.data;
    // Cập nhật nếu ID đã tồn tại, ngược lại thêm dòng mới
    var lastRow = lSheet.getLastRow();
    var updated = false;
    if (lastRow > 1) {
      var ids = lSheet.getRange(2, 1, lastRow - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(l.id)) {
          lSheet.getRange(i + 2, 1, 1, 6).setValues([[
            l.id, l.title, l.mediaUrl || "",
            JSON.stringify(l.paragraphs || []),
            JSON.stringify(l.timestamps || []),
            JSON.stringify(l.keywords || [])
          ]]);
          updated = true;
          break;
        }
      }
    }
    if (!updated) {
      lSheet.appendRow([
        l.id, l.title, l.mediaUrl || "",
        JSON.stringify(l.paragraphs || []),
        JSON.stringify(l.timestamps || []),
        JSON.stringify(l.keywords || [])
      ]);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Scores");
  var scores = [];
  if (sheet && sheet.getLastRow() > 1) {
    var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 10).getValues();
    scores = rows.map(function(row) {
      return {
        id: String(row[0]),
        dateISO: String(row[1]),
        dateVN: String(row[2]),
        lessonTitle: String(row[3]),
        scopeLabel: String(row[4]),
        modeLabel: String(row[5]),
        correct: Number(row[6]),
        total: Number(row[7]),
        percent: Number(row[8]),
        wrongWords: row[9] ? String(row[9]).split(",").map(function(s){ return s.trim(); }).filter(Boolean) : []
      };
    });
  }

  var lSheet = ss.getSheetByName("Lessons");
  var lessons = [];
  if (lSheet && lSheet.getLastRow() > 1) {
    var lRows = lSheet.getRange(2, 1, lSheet.getLastRow() - 1, 6).getValues();
    lessons = lRows.map(function(row) {
      return {
        id: String(row[0]),
        title: String(row[1]),
        mediaUrl: String(row[2] || ""),
        isSample: false,
        paragraphs: JSON.parse(row[3] || "[]"),
        timestamps: JSON.parse(row[4] || "[]"),
        keywords: JSON.parse(row[5] || "[]")
      };
    });
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: "ok",
    scores: scores,
    lessons: lessons
  })).setMimeType(ContentService.MimeType.JSON);
}
